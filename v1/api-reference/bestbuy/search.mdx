title: "Bestbuy Search"
api: "GET https://app.retailed.io/api/v1/scraper/bestbuy/search"
description: "Search for products on Bestbuy"

### Header

<ParamField header="x-api-key" type="string" required={true}>
    This parameter specifies the private key you'll need for Retailed access.
</ParamField>

### Parameters

<ParamField query="query" type="string" required={true} placeholder="macbook">
    Search query
</ParamField>

<ParamField query="page" type="number" required={false} placeholder="1">
    Page number
</ParamField>

<ParamField query="sort" type="string" required={false} placeholder="best_selling">
    Sort by `best_discount` or `best_selling`
</ParamField>
