---
title: "Chrono24 Finder Search"
api: "GET https://piloterr.com/api/v2/finder/chrono24-search"
description: "This endpoint allows users to search for attributes related to Chrono24."
---

#### Search Query

<ParamField body="query" type="string" required={true}>
  This parameter filters attributes based on a search string. The search is non-case-sensitive and returns all attributes whose names contain the specified string.
</ParamField>

### Response

<ResponseField name="list" type="array">
    <Expandable title="Filter Object" defaultOpen={true}>
        <ResponseField name="id" type="string">
            The unique identifier of the filter group (e.g., "dialColor", "dialNumbers")
        </ResponseField>
        <ResponseField name="input_type" type="string">
            The type of input for the filter (e.g., "Checkbox", "IllustrationCheckbox")
        </ResponseField>
        <ResponseField name="label" type="string">
            The display label for the filter group
        </ResponseField>
        <ResponseField name="searchable" type="boolean">
            Whether the filter group is searchable
        </ResponseField>
        <ResponseField name="values" type="array">
            <Expandable title="Filter Value" defaultOpen={true}>
                <ResponseField name="label" type="string">
                    The display label for the filter value
                </ResponseField>
                <ResponseField name="[filter_id]" type="string">
                    The value corresponding to the filter. The property name matches the filter's id
                    (e.g., dial_color: "702", dial_numbers: "801", other_attributes: "1201")
                </ResponseField>
            </Expandable>
        </ResponseField>
        <ResponseField name="filterGroupId" type="string">
            The group identifier for the filter (may match id)
        </ResponseField>
        <ResponseField name="inputType" type="string">
            Alternative property for input type (may match input_type)
        </ResponseField>
    </Expandable>
</ResponseField>
